-- ============================================================================
-- HỆ THỐNG CƠ SỞ DỮ LIỆU: BOOKING CARE (PHIÊN BẢN CHUẨN HÓA)
-- ============================================================================

CREATE DATABASE IF NOT EXISTS booking_care_db 
CHARACTER SET utf8mb4 
COLLATE utf8mb4_unicode_ci;

USE booking_care_db;

SET FOREIGN_KEY_CHECKS = 0;

-- Xóa bảng theo thứ tự ngược dependency
DROP TABLE IF EXISTS `articles`;
DROP TABLE IF EXISTS `refresh_tokens`;
DROP TABLE IF EXISTS `notifications`;
DROP TABLE IF EXISTS `reviews`;
DROP TABLE IF EXISTS `prescription_details`;
DROP TABLE IF EXISTS `prescriptions`;
DROP TABLE IF EXISTS `medicines`;
DROP TABLE IF EXISTS `medical_records`;
DROP TABLE IF EXISTS `invoices`;
DROP TABLE IF EXISTS `appointments`;
DROP TABLE IF EXISTS `schedules`;
DROP TABLE IF EXISTS `patient_profiles`;
DROP TABLE IF EXISTS `doctor_workplaces`;
DROP TABLE IF EXISTS `doctor_profiles`;
DROP TABLE IF EXISTS `specialties`;
DROP TABLE IF EXISTS `clinics`;
DROP TABLE IF EXISTS `users`;
DROP TABLE IF EXISTS `roles`;

SET FOREIGN_KEY_CHECKS = 1;

-- ============================================================================
-- 1. BẢNG ROLES
-- ============================================================================
CREATE TABLE roles (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Khóa chính tự tăng',
    name VARCHAR(50) NOT NULL UNIQUE COMMENT 'Tên vai trò (Admin, Doctor, Patient, Receptionist...)',
    description VARCHAR(255) COMMENT 'Mô tả vai trò',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB COMMENT='Bảng phân quyền hệ thống';

-- ============================================================================
-- 2. BẢNG USERS
-- ============================================================================
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Khóa chính',
    email VARCHAR(100) NOT NULL UNIQUE COMMENT 'Email đăng nhập',
    password VARCHAR(255) NOT NULL COMMENT 'Mật khẩu đã hash',
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    gender ENUM('Male','Female','Other'),
    date_of_birth DATE,
    address VARCHAR(255),
    avatar TEXT,
    role_id INT NOT NULL,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (role_id) REFERENCES roles(id) ON DELETE RESTRICT,
    INDEX idx_user_email (email),
    INDEX idx_user_phone (phone)
) ENGINE=InnoDB COMMENT='Bảng tài khoản người dùng';

-- ============================================================================
-- 3. BẢNG CLINICS
-- ============================================================================
CREATE TABLE clinics (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    address VARCHAR(255) NOT NULL,
    phone VARCHAR(20),
    email VARCHAR(100),
    description TEXT,
    image TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB COMMENT='Cơ sở y tế / Phòng khám';

-- ============================================================================
-- 4. BẢNG SPECIALTIES
-- ============================================================================
CREATE TABLE specialties (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    image TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB COMMENT='Chuyên khoa';

-- ============================================================================
-- 5. BẢNG DOCTOR_PROFILES
-- ============================================================================
CREATE TABLE doctor_profiles (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL UNIQUE,
    position VARCHAR(100),
    degree VARCHAR(100),
    experience_years INT DEFAULT 0,
    consultation_fee DECIMAL(10,2) DEFAULT 0.00,
    biography TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='Hồ sơ chuyên môn bác sĩ';

-- ============================================================================
-- 6. BẢNG DOCTOR_WORKPLACES
-- ============================================================================
CREATE TABLE doctor_workplaces (
    id INT AUTO_INCREMENT PRIMARY KEY,
    doctor_id INT NOT NULL,
    clinic_id INT NOT NULL,
    specialty_id INT NOT NULL,
    consultation_fee DECIMAL(10,2) DEFAULT 0.00,
    room VARCHAR(50),
    position VARCHAR(100),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (doctor_id) REFERENCES doctor_profiles(user_id) ON DELETE CASCADE,
    FOREIGN KEY (clinic_id) REFERENCES clinics(id) ON DELETE CASCADE,
    FOREIGN KEY (specialty_id) REFERENCES specialties(id) ON DELETE RESTRICT,
    
    UNIQUE KEY uk_doctor_workplace (doctor_id, clinic_id, specialty_id),
    INDEX idx_doctor_clinic (doctor_id, clinic_id)
) ENGINE=InnoDB COMMENT='Bác sĩ làm việc tại nhiều phòng khám';

-- ============================================================================
-- 7. BẢNG PATIENT_PROFILES
-- ============================================================================
CREATE TABLE patient_profiles (
    id INT AUTO_INCREMENT PRIMARY KEY,
    account_id INT NOT NULL,
    full_name VARCHAR(150) NOT NULL,
    phone VARCHAR(20),
    gender ENUM('Male','Female','Other') NOT NULL,
    date_of_birth DATE NOT NULL,
    relationship VARCHAR(50) DEFAULT 'Self',
    blood_type VARCHAR(5),
    height DECIMAL(5,2),
    weight DECIMAL(5,2),
    insurance_number VARCHAR(100),
    emergency_contact VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (account_id) REFERENCES users(id) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='Hồ sơ bệnh nhân (đặt lịch hộ)';

-- ============================================================================
-- 8. BẢNG SCHEDULES
-- ============================================================================
CREATE TABLE schedules (
    id INT AUTO_INCREMENT PRIMARY KEY,
    doctor_workplace_id INT NOT NULL,
    work_date DATE NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    max_patients INT DEFAULT 10,
    booked_patients INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    UNIQUE KEY uk_schedule (doctor_workplace_id, work_date, start_time),
    FOREIGN KEY (doctor_workplace_id) REFERENCES doctor_workplaces(id) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='Lịch làm việc của bác sĩ';

-- ============================================================================
-- 9. BẢNG APPOINTMENTS
-- ============================================================================
CREATE TABLE appointments (
    id INT AUTO_INCREMENT PRIMARY KEY,
    schedule_id INT NOT NULL,
    patient_profile_id INT NOT NULL,
    reason TEXT,
    status ENUM('PENDING','CONFIRMED','COMPLETED','CANCELLED') DEFAULT 'PENDING',
    booking_code VARCHAR(50) UNIQUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (schedule_id) REFERENCES schedules(id) ON DELETE RESTRICT,
    FOREIGN KEY (patient_profile_id) REFERENCES patient_profiles(id) ON DELETE RESTRICT
) ENGINE=InnoDB COMMENT='Lượt đặt lịch khám';

-- ============================================================================
-- 10. BẢNG INVOICES
-- ============================================================================
CREATE TABLE invoices (
    id INT AUTO_INCREMENT PRIMARY KEY,
    appointment_id INT NOT NULL,
    invoice_type ENUM('CLINIC_FEE', 'MEDICINE_FEE') NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    payment_method ENUM('CASH', 'VNPAY', 'MOMO', 'BANK_TRANSFER') DEFAULT 'CASH',
    payment_status ENUM('UNPAID', 'PAID', 'REFUNDED') DEFAULT 'UNPAID',
    transaction_id VARCHAR(100),
    payment_date DATETIME,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (appointment_id) REFERENCES appointments(id) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='Quản lý hóa đơn';

-- ============================================================================
-- 11. BẢNG MEDICAL_RECORDS
-- ============================================================================
CREATE TABLE medical_records (
    id INT AUTO_INCREMENT PRIMARY KEY,
    appointment_id INT NOT NULL UNIQUE,
    symptoms TEXT,
    diagnosis TEXT,
    conclusion TEXT,
    note TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (appointment_id) REFERENCES appointments(id) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='Bệnh án';

-- ============================================================================
-- 12. BẢNG MEDICINES
-- ============================================================================
CREATE TABLE medicines (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    unit VARCHAR(50),
    price DECIMAL(10,2),
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB COMMENT='Danh mục thuốc';

-- ============================================================================
-- 13. BẢNG PRESCRIPTIONS
-- ============================================================================
CREATE TABLE prescriptions (
    id INT AUTO_INCREMENT PRIMARY KEY,
    medical_record_id INT NOT NULL UNIQUE,
    note TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (medical_record_id) REFERENCES medical_records(id) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='Đơn thuốc tổng';

-- ============================================================================
-- 14. BẢNG PRESCRIPTION_DETAILS
-- ============================================================================
CREATE TABLE prescription_details (
    id INT AUTO_INCREMENT PRIMARY KEY,
    prescription_id INT NOT NULL,
    medicine_id INT NOT NULL,
    quantity INT NOT NULL,
    price_at_sale DECIMAL(10,2) NOT NULL,
    dosage VARCHAR(255),
    instruction TEXT,
    
    FOREIGN KEY (prescription_id) REFERENCES prescriptions(id) ON DELETE CASCADE,
    FOREIGN KEY (medicine_id) REFERENCES medicines(id)
) ENGINE=InnoDB COMMENT='Chi tiết đơn thuốc';

-- ============================================================================
-- 15. BẢNG REVIEWS
-- ============================================================================
CREATE TABLE reviews (
    id INT AUTO_INCREMENT PRIMARY KEY,
    doctor_id INT NOT NULL,
    patient_profile_id INT NOT NULL,
    appointment_id INT NOT NULL UNIQUE,
    rating TINYINT CHECK (rating BETWEEN 1 AND 5),
    comment TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (doctor_id) REFERENCES users(id),
    FOREIGN KEY (patient_profile_id) REFERENCES patient_profiles(id),
    FOREIGN KEY (appointment_id) REFERENCES appointments(id)
) ENGINE=InnoDB COMMENT='Đánh giá bác sĩ';

-- ============================================================================
-- 16. BẢNG NOTIFICATIONS
-- ============================================================================
CREATE TABLE notifications (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    title VARCHAR(255),
    content TEXT,
    is_read BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='Thông báo';

-- ============================================================================
-- 17. BẢNG REFRESH_TOKENS
-- ============================================================================
CREATE TABLE refresh_tokens (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    token TEXT NOT NULL,
    expires_at DATETIME NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='Refresh token JWT';

-- ============================================================================
-- 18. BẢNG ARTICLES
-- ============================================================================
CREATE TABLE articles (
    id INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    slug VARCHAR(255) UNIQUE,
    content_markdown LONGTEXT,
    content_html LONGTEXT,
    description TEXT,
    image TEXT,
    author_id INT,
    article_type ENUM('DOCTOR','CLINIC','SPECIALTY','NEWS') NOT NULL,
    reference_id INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (author_id) REFERENCES users(id) ON DELETE SET NULL
) ENGINE=InnoDB COMMENT='Bài viết / Cẩm nang';
